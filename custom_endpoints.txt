// run in console for recharging - update amount apne hisaab se

fetch('/api/wallet/dev-recharge', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer ' + localStorage.getItem('hc_access')
  },
  body: JSON.stringify({ amount: 10000 })
})
.then(res => res.json())
.then(data => console.log("Success! Wallet balance updated to:", data.data.balance));
